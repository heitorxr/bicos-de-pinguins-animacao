 93 | montar_transformacao <- function(modelo, progresso) {
 94 |   if (length(progresso) != 1L || !is.numeric(progresso) ||
 95 |       !is.finite(progresso) || progresso < 0 || progresso > 1) {
 96 |     stop("progresso deve ser um número entre 0 e 1.")
 97 |   }
 98 |   dados <- modelo$dados
 99 |   geral <- lm(bill_dep ~ bill_len, data = dados)
100 |   faixa <- range(dados$bill_len)
101 |   x_geral <- seq(faixa[1], faixa[2], length.out = 60)
102 |   y_geral <- as.numeric(predict(geral, newdata = data.frame(bill_len = x_geral)))
103 |   angulo_inicial <- atan(coef(geral)[[2]])
104 |   comprimento_inicial <- diff(faixa) * sqrt(1 + coef(geral)[[2]]^2) / 3
105 |   # As três partes começam coladas. O movimento cria a separação, sem afastamento prévio.
106 |   # A suavização faz o movimento começar e terminar devagar.
107 |   mover <- 3 * progresso^2 - 2 * progresso^3
108 |   grupos <- levels(dados$species)
109 |   # Ordena as espécies pelo centro da faixa de comprimentos, sem mover os pontos.
110 |   ordem <- dados |>
111 |     dplyr::group_by(species) |>
112 |     dplyr::summarise(centro_da_faixa = mean(range(bill_len)), .groups = "drop") |>
113 |     dplyr::arrange(centro_da_faixa) |>
114 |     dplyr::pull(species) |>
115 |     as.character()
116 |   # Mistura o cinza inicial com a cor da espécie usando o mesmo peso do giro.
117 |   cores <- grupos |>
118 |     purrr::set_names() |>
119 |     purrr::map_chr(function(grupo) {
120 |       rgb <- round((1 - mover) * grDevices::col2rgb(cores_pinguins[["Todos"]]) +
121 |         mover * grDevices::col2rgb(cores_pinguins[[grupo]]))
122 |       grDevices::rgb(rgb[1], rgb[2], rgb[3], maxColorValue = 255)
123 |     })
124 |   ramos <- purrr::map_dfr(grupos, function(grupo) {
125 |     dados_especie <- dplyr::filter(dados, species == grupo)
126 |     ajuste <- lm(bill_dep ~ bill_len, data = dados_especie)
127 |     x_final <- range(dados_especie$bill_len)
128 |     y_final <- as.numeric(predict(ajuste, newdata = tibble::tibble(bill_len = x_final)))
129 |     indice <- match(grupo, ordem)
130 |     centro_x <- faixa[1] + (indice - 0.5) * diff(faixa) / 3
131 |     centro_y <- as.numeric(predict(geral, newdata = data.frame(bill_len = centro_x)))
132 |     # Ângulo, comprimento, posição e cor usam exatamente o mesmo progresso.
133 |     angulo <- (1 - mover) * angulo_inicial + mover * atan2(diff(y_final), diff(x_final))
134 |     comprimento <- (1 - mover) * comprimento_inicial + mover * sqrt(diff(x_final)^2 + diff(y_final)^2)
135 |     centro_x <- (1 - mover) * centro_x + mover * mean(x_final)
136 |     centro_y <- (1 - mover) * centro_y + mover * mean(y_final)
137 |     posicao <- seq(-0.5, 0.5, length.out = 60)
138 |     cor_atual <- cores[[grupo]]
139 |     tibble::tibble(
140 |       x = centro_x + posicao * comprimento * cos(angulo),
141 |       y = centro_y + posicao * comprimento * sin(angulo),
142 |       grupo = grupo, alpha = 1, cor = cor_atual,
143 |       angulo = angulo, comprimento = comprimento, sincronia = mover
144 |     )
145 |   }) |> as.data.frame()
146 |   pontos <- dados |>
147 |     dplyr::mutate(grupo = as.character(species), cor = unname(cores[grupo])) |>
148 |     as.data.frame()
149 |   referencia <- tibble::tibble(x = x_geral, y = y_geral,
150 |                             alpha_solida = 0, alpha_tracejada = 0) |> as.data.frame()
151 |   intermediario <- progresso > 0 && progresso < 1
152 |   rotulo <- rotulo_correlacao(modelo, por_especie = progresso >= 0.5)
153 |   list(pontos = pontos, ramos = ramos, referencia = referencia,
154 |        alpha_ramos = 1, alpha_geral = 0, sincronia = mover,
155 |        etapa = if (progresso == 0) "unida" else "transformacao",
156 |        intermediario = intermediario, rotulo = rotulo)
157 | }
